// 0.1 Rich Text Editor Component
const RichTextEditor = ({ value, onChange, placeholder }) => {
  const contentRef = useRef(null);
  const [showColorPicker, setShowColorPicker] = useState(false);

  // 確保外部傳入新資料時，編輯器能正確同步
  useEffect(() => {
    if (contentRef.current && value !== contentRef.current.innerHTML) {
      contentRef.current.innerHTML = value || '';
    }
  }, [value]);

  const handleInput = () => {
    if (contentRef.current) {
      onChange(contentRef.current.innerHTML);
    }
  };

  const execCommand = (command, arg = null) => {
    document.execCommand(command, false, arg);
    handleInput(); 
    if (contentRef.current) contentRef.current.focus();
  };

  const colors = [
    { name: '黑色', value: '#334155' }, 
    { name: '紅色', value: '#ef4444' }, 
    { name: '橘色', value: '#f97316' }, 
    { name: '藍色', value: '#3b82f6' }, 
    { name: '綠色', value: '#22c55e' }, 
  ];

  return (
    <div className="flex flex-col border border-slate-200 rounded-2xl overflow-hidden bg-slate-50 focus-within:ring-4 focus-within:ring-orange-500/10 focus-within:border-orange-500/50 transition-all">
      {/* Toolbar */}
      <div className="flex items-center gap-1 p-2 border-b border-slate-200 bg-white">
        <button type="button" onMouseDown={(e) => { e.preventDefault(); execCommand('bold'); }} className="p-2 text-slate-600 hover:bg-slate-100 rounded-lg transition-colors" title="加粗"><Bold size={18} /></button>
        <button type="button" onMouseDown={(e) => { e.preventDefault(); execCommand('underline'); }} className="p-2 text-slate-600 hover:bg-slate-100 rounded-lg transition-colors" title="底線"><Underline size={18} /></button>
        <div className="relative">
          <button type="button" onMouseDown={(e) => { e.preventDefault(); setShowColorPicker(!showColorPicker); }} className={`p-2 text-slate-600 hover:bg-slate-100 rounded-lg transition-colors ${showColorPicker ? 'bg-slate-100' : ''}`} title="文字顏色"><Palette size={18} /></button>
          {showColorPicker && (
            <div className="absolute top-full left-0 mt-1 bg-white border border-slate-200 shadow-xl rounded-lg p-2 flex gap-1 z-20">
              {colors.map((c) => (
                <button key={c.value} type="button" onMouseDown={(e) => { e.preventDefault(); execCommand('foreColor', c.value); setShowColorPicker(false); }} className="w-6 h-6 rounded-full border border-slate-100 hover:scale-110 transition-transform" style={{ backgroundColor: c.value }} title={c.name} />
              ))}
            </div>
          )}
        </div>
        <div className="w-px h-5 bg-slate-300 mx-1"></div>
        <button type="button" onMouseDown={(e) => { e.preventDefault(); execCommand('insertOrderedList'); }} className="p-2 text-slate-600 hover:bg-slate-100 rounded-lg transition-colors" title="編號清單"><ListOrdered size={18} /></button>
        <button type="button" onMouseDown={(e) => { e.preventDefault(); execCommand('insertUnorderedList'); }} className="p-2 text-slate-600 hover:bg-slate-100 rounded-lg transition-colors" title="符號清單"><List size={18} /></button>
      </div>

      {/* Editor Area */}
      <div
        ref={contentRef}
        contentEditable
        onInput={handleInput}
        className="w-full p-5 min-h-[150px] outline-none text-slate-700 font-medium leading-relaxed resize-y overflow-auto text-lg list-inside-custom"
        style={{ maxHeight: '400px' }}
      />
      {!value && (
        <div 
          className="absolute top-[60px] left-5 text-slate-400 font-medium pointer-events-none"
          onClick={() => contentRef.current?.focus()}
        >
          {placeholder}
        </div>
      )}
      
      <style>{`
        .list-inside-custom ul { list-style-type: disc; padding-left: 1.5em; margin-bottom: 0.6em; }
        .list-inside-custom ol { list-style-type: decimal; padding-left: 1.5em; margin-bottom: 0.6em; }
        .list-inside-custom li { margin-bottom: 0.5em; } 
      `}</style>
    </div>
  );
};

// 0.2 Formatted Text Component
const FormattedText = ({ text, className }) => {
  if (!text) return null;
  const lines = text.split('\n');
  return (
    <div className={className}>
      {lines.map((line, index) => {
        const listMatch = line.match(/^(\d+\.|[•-])\s+(.*)/);
        if (listMatch) {
          const marker = listMatch[1];
          const content = listMatch[2];
          return (
            <div key={index} className="flex items-start gap-2 mb-5 last:mb-0">
              <span className="shrink-0 font-bold text-slate-800 w-6 text-right mr-1">{marker}</span>
              <div className="flex-1 min-w-0 leading-relaxed text-justify">{content}</div>
            </div>
          );
        }
        if (!line.trim()) return <div key={index} className="h-3" />; 
        return <div key={index} className="mb-2 leading-relaxed">{line}</div>;
      })}
    </div>
  );
};

// 0.3 Display Component
const NoteContentDisplay = ({ content, isExpanded }) => {
  if (!content) return null;
  const isRichText = /<[a-z][\s\S]*>/i.test(content);

  if (isRichText) {
    return (
      <div 
        className={`text-slate-600 text-lg font-medium leading-relaxed rich-text-content ${!isExpanded ? 'line-clamp-3' : ''}`}
        dangerouslySetInnerHTML={{ __html: content }}
      />
    );
  }
  return (
    <div className={`text-slate-600 text-lg font-medium ${!isExpanded ? 'line-clamp-3' : ''}`}>
      <FormattedText text={content} />
    </div>
  );
};

// 2.5 View Note Modal Component (負責呈現 Popup，並限定在內容區域)
const ViewNoteModal = ({ note, categoryLabel, onClose, onEdit, onDelete }) => {
  const [isClosing, setIsClosing] = useState(false);

  // 當開啟新的筆記時，重置關閉狀態
  useEffect(() => {
    if (note) setIsClosing(false);
  }, [note]);

  if (!note) return null;
  const displayTitle = note.title || note.question;
  const displayOverview = note.title ? note.question : null;

  // 處理關閉動畫：先觸發 isClosing 狀態，300毫秒後再實際執行關閉
  const handleCloseAnim = () => {
    if (isClosing) return;
    setIsClosing(true);
    setTimeout(() => onClose(), 300);
  };

  const handleEditAnim = () => {
    if (isClosing) return;
    setIsClosing(true);
    setTimeout(() => onEdit(note), 300);
  };

  const handleDeleteAnim = () => {
    if (isClosing) return;
    setIsClosing(true);
    setTimeout(() => onDelete(note.id), 300);
  };

  return (
    <div 
      className={`absolute inset-0 bg-black/25 z-[60] flex items-center justify-center p-4 sm:p-6 backdrop-blur-sm ${isClosing ? 'animate-fade-exit' : 'animate-fade-enter'}`}
      onClick={(e) => { e.stopPropagation(); handleCloseAnim(); }}
    >
      {/* 外層容器：加入 transform-gpu 與 will-change 確保硬體加速，避免縮放時閃爍 */}
      <div 
        className={`bg-white rounded-3xl w-full max-w-xl sm:max-w-2xl max-h-[95%] shadow-2xl flex flex-col relative origin-center transform-gpu will-change-[transform,opacity] ${isClosing ? 'animate-modal-exit' : 'animate-modal-enter'}`}
        onClick={(e) => e.stopPropagation()}
      >
        {/* 右上角固定關閉按鈕 */}
        <button 
          onClick={handleCloseAnim} 
          className="absolute top-4 right-4 z-[70] p-2 bg-white/80 hover:bg-white text-slate-800 rounded-full backdrop-blur-md shadow-md border border-slate-200 transition-transform hover:scale-110"
          title="關閉"
        >
          <X size={20} />
        </button>

        {/* 內層捲動區域：包含圖片與文字內容。加入 overscroll-contain 避免在底部滑動時牽連背景 */}
        <div className="overflow-y-auto overscroll-contain custom-scrollbar flex flex-col w-full h-full rounded-3xl">
          {/* 放大後的等比例圖卡 */}
          {note.imageUrl && (
            <div className="w-full h-56 sm:h-72 shrink-0 relative bg-slate-100">
              <img src={note.imageUrl} alt="Reference" className="w-full h-full object-cover" onError={(e) => { e.target.onerror = null; e.target.src = "https://via.placeholder.com/800x450?text=No+Image"; }} />
              <div className="absolute top-4 left-4 z-20">
                 <span className="px-3 py-1.5 rounded-full text-xs font-bold bg-white/95 text-slate-800 shadow-lg backdrop-blur-md border border-white/50 tracking-wide">{categoryLabel}</span>
              </div>
            </div>
          )}

          {/* Popup 詳細內容區 */}
          <div className="px-7 pt-5 pb-6 flex flex-col flex-1 relative z-0">
            {!note.imageUrl && (
              <div className="mb-5"><span className="px-3 py-1.5 rounded-full text-xs font-bold bg-slate-100/80 text-slate-600 border border-slate-200 tracking-wide">{categoryLabel}</span></div>
            )}
            
            <h3 className="font-bold text-xl text-slate-800 mb-2 leading-relaxed tracking-tight">
              {displayTitle}
            </h3>
            
            {displayOverview && (
              <div className="mb-4 text-slate-500 text-sm leading-relaxed border-l-2 border-slate-300 pl-3">
                {displayOverview}
              </div>
            )}

            <div className="mb-5">
              <div className="font-bold text-base text-slate-700 mb-3 flex items-center gap-2">
                <div className="p-1.5 rounded-lg bg-amber-100 text-amber-600"><Lightbulb size={16} fill="currentColor" /></div>
                專家建議：
              </div>
              <NoteContentDisplay content={note.answer} isExpanded={true} />
            </div>

            {/* 彈出視窗底層的按鈕區 */}
            <div className="mt-auto pt-3 border-t border-slate-200/60 flex items-center justify-between">

// 6. Data Management Modal (Backup/Restore)
const DataManagementModal = ({ isOpen, onClose, notes, categories }) => {
  const [activeTab, setActiveTab] = useState('export'); // 'export' or 'import'
  const [importData, setImportData] = useState('');
  const [statusMsg, setStatusMsg] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  if (!isOpen) return null;

  const getExportData = () => {
    return JSON.stringify({
      version: 1,
      timestamp: new Date().toISOString(),
      categories: categories,
      notes: notes
    }, null, 2);
  };

  const handleCopy = () => {
    const textToCopy = getExportData();
    // Fallback using document.execCommand('copy')
    const textarea = document.createElement('textarea');
    textarea.value = textToCopy;
    textarea.style.position = 'fixed';  // Prevent scrolling to bottom of page in MS Edge.
    document.body.appendChild(textarea);
    textarea.select();
    try {
      document.execCommand('copy');
      setStatusMsg('✅ 資料已複製到剪貼簿！');
      setTimeout(() => setStatusMsg(''), 3000);
    } catch (err) {
      console.error('Fallback: Oops, unable to copy', err);
      setStatusMsg('❌ 複製失敗，請手動選取複製');
    }
    document.body.removeChild(textarea);
  };

  const handleImport = async () => {
    if (!importData.trim()) return;
    setIsLoading(true);
    setStatusMsg('');

    try {
      const data = JSON.parse(importData);
      if (!data.notes && !data.categories) {
        throw new Error("無效的資料格式");
      }

      const batch = writeBatch(db);
      const categoryIdMap = {};

      if (data.categories) {
        data.categories.forEach(cat => {
          const oldId = cat.id;
          const newCatRef = doc(getPublicCollection('categories')); 
          const { id, ...catData } = cat; 
          categoryIdMap[oldId] = newCatRef.id;
          batch.set(newCatRef, { ...catData, createdAt: serverTimestamp() });
        });
      }
      
      if (data.notes) {
        data.notes.forEach(note => {
          const newNoteRef = doc(getPublicCollection('notes'));
          const { id, ...noteData } = note;
          // Update category ID to new ID if exists
          if (noteData.category && categoryIdMap[noteData.category]) {
            noteData.category = categoryIdMap[noteData.category];
          }
          batch.set(newNoteRef, { ...noteData, timestamp: serverTimestamp() });
        });
      }

      await batch.commit();
      setStatusMsg(`✅ 成功還原 ${data.notes?.length || 0} 筆筆記！`);
      setImportData('');
      setTimeout(() => {
        onClose();
        setStatusMsg('');
      }, 1500);

    } catch (e) {
      console.error(e);
      setStatusMsg('❌ 還原失敗：資料格式錯誤');
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="fixed inset-0 bg-slate-900/40 backdrop-blur-md z-[80] flex items-center justify-center p-4">
      <div className="bg-white rounded-3xl w-full max-w-2xl shadow-2xl border border-white/50 overflow-hidden flex flex-col max-h-[80vh]">
        <div className="p-6 border-b border-slate-100 flex justify-between items-center bg-slate-50/50">
          <h2 className="text-xl font-bold text-slate-800 flex items-center gap-2">
            <Database size={20} className="text-blue-600" />
            資料備份與還原
          </h2>
          <button onClick={onClose} className="p-2 hover:bg-slate-200 rounded-full transition-colors"><X size={20} /></button>
        </div>
        
        <div className="flex border-b border-slate-200">
          <button onClick={() => setActiveTab('export')} className={`flex-1 py-4 font-bold text-sm flex items-center justify-center gap-2 transition-colors ${activeTab === 'export' ? 'bg-white text-blue-600 border-b-2 border-blue-600' : 'bg-slate-50 text-slate-500 hover:bg-slate-100'}`}>
            <Download size={16} /> 匯出備份 (Export)
          </button>
          <button onClick={() => setActiveTab('import')} className={`flex-1 py-4 font-bold text-sm flex items-center justify-center gap-2 transition-colors ${activeTab === 'import' ? 'bg-white text-orange-600 border-b-2 border-orange-600' : 'bg-slate-50 text-slate-500 hover:bg-slate-100'}`}>
            <Upload size={16} /> 還原資料 (Import)
          </button>
        </div>

        <div className="p-6 flex-1 overflow-y-auto">
          {activeTab === 'export' ? (
            <div className="space-y-4">
              <p className="text-sm text-slate-500">如果程式碼更新導致資料消失，請先複製此處的資料代碼，待更新完成後再使用「還原功能」貼上。</p>
              <div className="relative">
                <textarea readOnly value={getExportData()} className="w-full h-64 p-4 bg-slate-800 text-green-400 font-mono text-xs rounded-xl focus:outline-none resize-none custom-scrollbar" />
                <button onClick={handleCopy} className="absolute top-4 right-4 bg-white/10 hover:bg-white/20 text-white p-2 rounded-lg backdrop-blur-sm transition-colors" title="複製到剪貼簿"><Copy size={16} /></button>
              </div>
              <div className="flex justify-center">
                <button onClick={handleCopy} className="bg-blue-600 hover:bg-blue-700 text-white px-6 py-2 rounded-xl font-bold transition-colors flex items-center gap-2"><Copy size={18} /> 複製全部資料代碼</button>
              </div>
            </div>
          ) : (
            <div className="space-y-4">
              <p className="text-sm text-slate-500">請將之前備份的資料代碼貼在下方，點擊還原即可恢復分類與筆記。</p>
              <textarea placeholder='在此貼上 JSON 資料...' value={importData} onChange={(e) => setImportData(e.target.value)} className="w-full h-64 p-4 bg-slate-50 border border-slate-200 text-slate-700 font-mono text-xs rounded-xl focus:ring-2 focus:ring-orange-500/50 outline-none resize-none custom-scrollbar" />
              <div className="flex justify-center">
                <button onClick={handleImport} disabled={isLoading || !importData} className="bg-orange-500 hover:bg-orange-600 text-white px-6 py-2 rounded-xl font-bold transition-colors flex items-center gap-2 disabled:opacity-50">
                  {isLoading ? <div className="animate-spin h-4 w-4 border-2 border-white border-t-transparent rounded-full"/> : <Upload size={18} />}
                  開始還原資料
                </button>
              </div>
            </div>
          )}
          {statusMsg && <div className="mt-4 p-3 bg-slate-100 text-slate-800 text-center font-bold rounded-xl animate-bounce">{statusMsg}</div>}
        </div>
      </div>
    </div>
  );
};

      setCategories(catData);
    }, (error) => console.error("Error fetching categories:", error));

    return () => { unsubNotes(); unsubCats(); };
  }, [user]);

  const handleAddCategory = async (name) => {
    if (!user) return;
    try { await addDoc(getPublicCollection('categories'), { label: name, createdAt: serverTimestamp() }); } catch (error) { console.error("Error adding category:", error); }
  };

  const handleDeleteCategoryRequest = (catId) => { setDeleteState({ isOpen: true, type: 'category', id: catId, message: "分類下的筆記不會被刪除，但會失去分類標籤。" }); };
  const handleDeleteNoteRequest = (noteId) => { setDeleteState({ isOpen: true, type: 'note', id: noteId, message: "刪除後無法復原，您確定嗎？" }); };

  const confirmDelete = async () => {
    if (!user || !deleteState.id) return;
    try {
      if (deleteState.type === 'category') {
        await deleteDoc(getPublicDoc('categories', deleteState.id));
        if (activeCategory === deleteState.id) setActiveCategory('all');
      } else if (deleteState.type === 'note') {
        await deleteDoc(getPublicDoc('notes', deleteState.id));
      }
    } catch (error) { console.error("Error deleting item:", error); } finally { setDeleteState({ isOpen: false, type: null, id: null, message: '' }); }
  };

  const handleEditNote = (note) => { setEditingNote(note); setIsAddModalOpen(true); };
  const handleAddNote = () => { setEditingNote(null); setIsAddModalOpen(true); };
  const handleCreateCategoryFirst = () => { setIsSidebarOpen(true); };

  return (
    <div className="flex h-screen w-full bg-slate-50 text-slate-900" style={{ fontFamily: "'Noto Sans TC', sans-serif" }}>
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Noto+Sans+TC:wght@400;500;700&display=swap');
        .custom-scrollbar::-webkit-scrollbar { width: 6px; }
        .custom-scrollbar::-webkit-scrollbar-track { background: transparent; }
        .custom-scrollbar::-webkit-scrollbar-thumb { background-color: rgba(156, 163, 175, 0.5); border-radius: 20px; }
        
        /* 恢復富文本內容中的列表圓點與數字樣式 */
        .rich-text-content ul { list-style-type: disc; padding-left: 1.2em; margin-bottom: 0.6em; }
        .rich-text-content ol { list-style-type: decimal; padding-left: 1.2em; margin-bottom: 0.6em; }
        .rich-text-content li { margin-bottom: 0.5em; }

        /* Popup 動畫效果 (使用 both 確保不會有任何一瞬間的閃爍，並加上 will-change 硬體加速) */
        @keyframes modalEnter {
          from { opacity: 0; transform: scale(0.65) translateY(20px); }
          to { opacity: 1; transform: scale(1) translateY(0); }
        }
        @keyframes modalExit {
          from { opacity: 1; transform: scale(1) translateY(0); }
          to { opacity: 0; transform: scale(0.65) translateY(20px); }
        }
        @keyframes fadeEnter {
          from { opacity: 0; }
          to { opacity: 1; }
        }
        @keyframes fadeExit {
          from { opacity: 1; }
          to { opacity: 0; }
        }
        
        .animate-modal-enter { 
          animation: modalEnter 0.4s cubic-bezier(0.16, 1, 0.3, 1) both; 
          will-change: transform, opacity;
        }
        .animate-modal-exit { 
          animation: modalExit 0.3s cubic-bezier(0.16, 1, 0.3, 1) both; 
          will-change: transform, opacity;
        }
        .animate-fade-enter { animation: fadeEnter 0.3s ease-out both; }
        .animate-fade-exit { animation: fadeExit 0.3s ease-in both; }
      `}</style>

      {/* Mobile Menu & Add Buttons Container */}
      <div className="md:hidden fixed top-7 right-4 z-40 flex flex-col gap-4 items-center">
        <button 
          onClick={() => setIsSidebarOpen(!isSidebarOpen)}
          className="bg-white/80 backdrop-blur-md p-2 rounded-lg shadow-md text-slate-700 border border-white/40 hover:bg-white transition-colors"
        >
          <Menu />
        </button>
        <button 
          onClick={handleAddNote}
          className="bg-orange-500 hover:bg-orange-600 text-white p-3 rounded-full shadow-lg shadow-orange-500/20 hover:shadow-orange-500/30 transition-all hover:-translate-y-0.5 active:scale-95"
        >
          <Plus size={24} />
        </button>
      </div>

      <Sidebar 
        activeCategory={activeCategory} 
        setActiveCategory={setActiveCategory}
        categories={categories}
        isOpen={isSidebarOpen}
        setIsOpen={setIsSidebarOpen}
        onAddCategory={handleAddCategory}
        onDeleteCategory={handleDeleteCategoryRequest}
        onOpenDataManagement={() => setIsDataModalOpen(true)}
      />

      <MainContent 
        user={user}
        notes={notes}
        categories={categories}
        loading={loading}
        activeCategory={activeCategory}
        onAddNote={handleAddNote}
        onEditNote={handleEditNote}
        onDeleteNote={handleDeleteNoteRequest}
      />

      <AddNoteModal 
        isOpen={isAddModalOpen} 
        onClose={() => setIsAddModalOpen(false)}
        user={user}
        appId={appId}
        db={db}
        editingNote={editingNote}
        categories={categories}
        onCreateCategoryFirst={handleCreateCategoryFirst}
      />

      <ConfirmModal 
        isOpen={deleteState.isOpen}
        onClose={() => setDeleteState({ ...deleteState, isOpen: false })}
        onConfirm={confirmDelete}
        message={deleteState.message}
      />

      <DataManagementModal 
        isOpen={isDataModalOpen}
        onClose={() => setIsDataModalOpen(false)}
        notes={notes}
        categories={categories}
      />
    </div>
  );
}
